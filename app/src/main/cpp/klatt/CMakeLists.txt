cmake_minimum_required(VERSION 3.22.1)
project(klatt_subbuild C)
set(CMAKE_C_STANDARD 99)
set(KLATT_ROOT ${CMAKE_CURRENT_LIST_DIR}/upstream)
add_library(klatt_core STATIC ${KLATT_ROOT}/parwave.c)
target_include_directories(klatt_core PUBLIC ${KLATT_ROOT})
set_target_properties(klatt_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_compile_definitions(klatt_core PRIVATE _DEFAULT_SOURCE=1)
target_compile_options(klatt_core PRIVATE -Wno-unused-parameter -Wno-sign-compare)
if(ANDROID)
 add_library(klattretro SHARED klatt_native.c)
 target_include_directories(klattretro PRIVATE ${CMAKE_CURRENT_LIST_DIR}/../../../../../flite/include ${CMAKE_CURRENT_LIST_DIR}/../../../../../flite/lang/usenglish ${CMAKE_CURRENT_LIST_DIR}/../../../../../flite/lang/cmulex ${KLATT_ROOT})
 target_link_libraries(klattretro klatt_core fliteretro m)
 target_link_options(klattretro PRIVATE -Wl,-z,max-page-size=16384 -Wl,-z,common-page-size=16384)
endif()
if(NOT ANDROID)
 add_executable(klatt_host_test klatt_host_test.c)
 target_include_directories(klatt_host_test PRIVATE ${KLATT_ROOT})
 target_link_libraries(klatt_host_test klatt_core m)
endif()
