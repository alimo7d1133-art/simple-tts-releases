cmake_minimum_required(VERSION 3.22.1)
project(mbrola_subbuild C)
set(CMAKE_C_STANDARD 99)
# Add with: add_subdirectory(mbrola) after fliteretro exists. This file is intentionally isolated.
set(MBROLA_ROOT ${CMAKE_CURRENT_LIST_DIR}/upstream)
add_library(mbrola_core STATIC ${MBROLA_ROOT}/mbrola_lib1.c)
target_include_directories(mbrola_core PUBLIC ${MBROLA_ROOT} ${MBROLA_ROOT}/Parser ${MBROLA_ROOT}/Engine ${MBROLA_ROOT}/Database ${MBROLA_ROOT}/Misc ${MBROLA_ROOT}/LibOneChannel)
set_target_properties(mbrola_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_compile_definitions(mbrola_core PRIVATE LIBRARY ROMDATABASE_STORE ROMDATABASE_INIT LITTLE_ENDIAN)
target_compile_options(mbrola_core PRIVATE -Wno-error=implicit-function-declaration -Wno-error=incompatible-pointer-types -fno-strict-aliasing)
if(ANDROID)
 add_library(mbrolaretro SHARED mbrola_native.c)
 target_include_directories(mbrolaretro PRIVATE ${CMAKE_CURRENT_LIST_DIR}/../../../../../flite/include ${CMAKE_CURRENT_LIST_DIR}/../../../../../flite/lang/usenglish ${CMAKE_CURRENT_LIST_DIR}/../../../../../flite/lang/cmulex ${MBROLA_ROOT}/Parser ${MBROLA_ROOT}/Engine ${MBROLA_ROOT}/Database ${MBROLA_ROOT}/Misc ${MBROLA_ROOT}/LibOneChannel)
 target_link_libraries(mbrolaretro mbrola_core fliteretro m)
 target_link_options(mbrolaretro PRIVATE -Wl,-z,max-page-size=16384 -Wl,-z,common-page-size=16384)
endif()

if(NOT ANDROID)
 add_executable(mbrola_host_test mbrola_host_test.c)
 target_include_directories(mbrola_host_test PRIVATE ${MBROLA_ROOT}/Parser ${MBROLA_ROOT}/Engine ${MBROLA_ROOT}/Database ${MBROLA_ROOT}/Misc ${MBROLA_ROOT}/LibOneChannel)
 target_link_libraries(mbrola_host_test mbrola_core m)
endif()
