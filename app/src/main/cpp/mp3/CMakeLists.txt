cmake_minimum_required(VERSION 3.22.1)
project(mp3encoder C)
set(LAME_DIR ${CMAKE_CURRENT_LIST_DIR}/lame)
file(GLOB LAME_SOURCES CONFIGURE_DEPENDS ${LAME_DIR}/libmp3lame/*.c)
list(REMOVE_ITEM LAME_SOURCES ${LAME_DIR}/libmp3lame/mpglib_interface.c)
add_library(mp3encoder SHARED mp3_encoder.c ${LAME_SOURCES})
target_include_directories(mp3encoder PRIVATE ${LAME_DIR} ${LAME_DIR}/include ${LAME_DIR}/libmp3lame)
target_compile_definitions(mp3encoder PRIVATE HAVE_CONFIG_H LAME_LIBRARY_BUILD)
target_compile_options(mp3encoder PRIVATE -Wno-error=implicit-function-declaration -Wno-error=incompatible-pointer-types -fno-strict-aliasing)
target_link_libraries(mp3encoder m)
if(ANDROID)
  target_link_options(mp3encoder PRIVATE -Wl,-z,max-page-size=16384 -Wl,-z,common-page-size=16384)
endif()
